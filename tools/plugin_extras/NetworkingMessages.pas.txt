// Hand-written additions to the generated ISteamNetworkingMessages plug-in.
// Merged verbatim by tools/gen_steam_api.py (sections start with //@name).
// Received messages must be released: see NetworkingSockets.pas.txt.
//@private
    { SteamAPI_SteamNetworkingMessage_t_Release: gives a received message
      back to Steam (a struct function of the flat API, bound by hand). }
    F_MessageRelease: procedure(AMessage: PSteamNetworkingMessage_t); cdecl;
//@public
    { Gives a message from @link(ReceiveMessages) (or the raw
      ReceiveMessagesOnChannel) back to Steam. Call it once for every
      message, after reading m_pData; @nil is ignored. }
    procedure ReleaseMessage(AMessage: PSteamNetworkingMessage_t);
    { Up to Length(AMessages) waiting messages on a local channel, into
      AMessages from index 0. @returns(How many were stored.) Release each
      with @link(ReleaseMessage). }
    function ReceiveMessages(nLocalChannel: Integer; var AMessages: array of PSteamNetworkingMessage_t): Integer;
//@initialize
  F_MessageRelease := nil;
  // optional like every other export; ReleaseMessage raises when it is missing
  Bind(F_MessageRelease, 'SteamAPI_SteamNetworkingMessage_t_Release', False);
//@implementation
procedure TSteamNetworkingMessagesPlugin.ReleaseMessage(AMessage: PSteamNetworkingMessage_t);
begin
  if AMessage = nil then
    Exit;
  // a silent no-op here would leak every message
  if not Assigned(F_MessageRelease) then
    raise ESteamworksError.Create('ReleaseMessage: the loaded Steam library does not export ' +
      'SteamAPI_SteamNetworkingMessage_t_Release');
  F_MessageRelease(AMessage);
end;

function TSteamNetworkingMessagesPlugin.ReceiveMessages(nLocalChannel: Integer;
  var AMessages: array of PSteamNetworkingMessage_t): Integer;
begin
  if Length(AMessages) = 0 then
    Exit(0);
  Result := ReceiveMessagesOnChannel(nLocalChannel, @AMessages[0], Length(AMessages));
end;
