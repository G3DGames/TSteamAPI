// Hand-written additions to the generated ISteamInput plug-in.
// Merged verbatim by tools/gen_steam_api.py (sections start with //@name).
// Everything here must follow the source rules (ASCII, PasDoc on every
// declaration in the interface sections).
//
// THE BY-VALUE SHIM. GetDigitalActionData, GetAnalogActionData and
// GetMotionData return a struct BY VALUE (all three under #pragma pack(1):
// 2, 13 and 40 bytes). The generator refuses that, because how C returns a
// struct depends on its size and the platform, and Pascal compilers do not
// always match (Delphi Win32 cdecl puts the hidden result pointer in a
// different place from MSVC). Each export is therefore declared to match
// the C ABI exactly, as clang lowers it (tools/verify_generated.py checks
// the register / hidden-pointer choice for every target on every SDK
// update; who pops the hidden pointer on 32-bit x86 was checked by hand
// with clang -S, see below):
//
//   export   Win32 / Win64      x64 Unix (Linux,     ARM64 (macOS,
//                               macOS x64)           Android, Linux)
//   Digital  i16 in AX          i16 in AX            i16 in W0
//   Analog   hidden pointer     RAX:RDX              X0:X1
//   Motion   hidden pointer     hidden pointer RDI   hidden pointer X8
//
// - "hidden pointer" on Windows and x64 Unix is the first argument, so it
//   is declared as an explicit first parameter. MSVC Win32: the CALLER pops
//   it (clang emits "retl"), which cdecl does.
// - Two-register returns are declared as a 16-byte record of two UInt64,
//   which both compilers return in the same two registers; the first 13
//   bytes are the struct.
// - ARM64 passes the hidden pointer in X8, not as an argument: there the
//   compiler's own record return is used (it implements AAPCS64).
// - Linux32 (FPC only, may be dropped): i386 System V returns EVERY struct
//   through a hidden pointer that the CALLEE pops ("retl $4"), which an
//   explicit cdecl parameter would unbalance; FPC's own record return is
//   used there. UNVERIFIED.
//@types
  { Controller handles, as returned by @link(TSteamInputPlugin.ConnectedControllers). }
  TSteamInputHandleArray = array of InputHandle_t;
  { Action origins (buttons / axes bound to an action), as returned by
    @link(TSteamInputPlugin.DigitalActionOrigins). }
  TSteamInputOriginArray = array of EInputActionOrigin;

const
  { Most controllers ISteamInput reports at once (C STEAM_INPUT_MAX_COUNT). }
  STEAM_INPUT_MAX_COUNT = 16;
  { Most origins one action can have (C STEAM_INPUT_MAX_ORIGINS). }
  STEAM_INPUT_MAX_ORIGINS = 8;

type

{$if defined(STEAMWORKS_CPU386) and not defined(STEAMWORKS_WINDOWS)}
  { Linux32: @code(SteamAPI_ISteamInput_GetDigitalActionData), compiler record return (unverified). }
  TSteamInputDigitalFn = function(AIntf: Pointer; inputHandle: InputHandle_t;
    digitalActionHandle: InputDigitalActionHandle_t): InputDigitalActionData_t; cdecl;
  { Linux32: @code(SteamAPI_ISteamInput_GetAnalogActionData), compiler record return (unverified). }
  TSteamInputAnalogFn = function(AIntf: Pointer; inputHandle: InputHandle_t;
    analogActionHandle: InputAnalogActionHandle_t): InputAnalogActionData_t; cdecl;
  { Linux32: @code(SteamAPI_ISteamInput_GetMotionData), compiler record return (unverified). }
  TSteamInputMotionFn = function(AIntf: Pointer; inputHandle: InputHandle_t): InputMotionData_t; cdecl;
{$else}
  { @code(SteamAPI_ISteamInput_GetDigitalActionData): the 2-byte struct
    comes back in AX / W0, so it is read as UInt16. }
  TSteamInputDigitalFn = function(AIntf: Pointer; inputHandle: InputHandle_t;
    digitalActionHandle: InputDigitalActionHandle_t): UInt16; cdecl;
  {$ifdef STEAMWORKS_WINDOWS}
  { @code(SteamAPI_ISteamInput_GetAnalogActionData) on Windows: hidden result
    pointer first (MSVC). }
  TSteamInputAnalogFn = procedure(AResult: PInputAnalogActionData_t; AIntf: Pointer;
    inputHandle: InputHandle_t; analogActionHandle: InputAnalogActionHandle_t); cdecl;
  { @code(SteamAPI_ISteamInput_GetMotionData) on Windows: hidden result pointer first (MSVC). }
  TSteamInputMotionFn = procedure(AResult: PInputMotionData_t; AIntf: Pointer;
    inputHandle: InputHandle_t); cdecl;
  {$else}
  { The 13-byte InputAnalogActionData_t as x64 / ARM64 Unix return it: two
    64-bit registers (RAX:RDX, X0:X1). }
  TSteamInputRet16 = record
    { First register: eMode and x. }
    Lo: UInt64;
    { Second register: y and bActive (low 5 bytes). }
    Hi: UInt64;
  end;
  { @code(SteamAPI_ISteamInput_GetAnalogActionData) on x64 / ARM64 Unix. }
  TSteamInputAnalogFn = function(AIntf: Pointer; inputHandle: InputHandle_t;
    analogActionHandle: InputAnalogActionHandle_t): TSteamInputRet16; cdecl;
    {$ifdef STEAMWORKS_CPUARM64}
  { @code(SteamAPI_ISteamInput_GetMotionData) on ARM64: hidden pointer in X8,
    which only the compiler's record return provides. }
  TSteamInputMotionFn = function(AIntf: Pointer; inputHandle: InputHandle_t): InputMotionData_t; cdecl;
    {$else}
  { @code(SteamAPI_ISteamInput_GetMotionData) on x64 Unix: hidden result
    pointer first (RDI). }
  TSteamInputMotionFn = procedure(AResult: PInputMotionData_t; AIntf: Pointer;
    inputHandle: InputHandle_t); cdecl;
    {$endif}
  {$endif}
{$endif}

//@private
    { Bound @code(SteamAPI_ISteamInput_GetDigitalActionData) (by-value shim). }
    F_GetDigitalActionData: TSteamInputDigitalFn;
    { Bound @code(SteamAPI_ISteamInput_GetAnalogActionData) (by-value shim). }
    F_GetAnalogActionData: TSteamInputAnalogFn;
    { Bound @code(SteamAPI_ISteamInput_GetMotionData) (by-value shim). }
    F_GetMotionData: TSteamInputMotionFn;
    { @true between a successful @link(Start) and @link(Stop). }
    FStarted: Boolean;
//@public
    { Current state of a digital action (a button) on one controller:
      @code(bState) pressed, @code(bActive) the action is in the active
      action set and bound. Poll it every frame. }
    function GetDigitalActionData(inputHandle: InputHandle_t;
      digitalActionHandle: InputDigitalActionHandle_t): InputDigitalActionData_t;
    { Current state of an analog action (a stick, trigger or pad) on one
      controller: @code(x) / @code(y), @code(eMode) the source,
      @code(bActive) as for digital actions. Poll it every frame. }
    function GetAnalogActionData(inputHandle: InputHandle_t;
      analogActionHandle: InputAnalogActionHandle_t): InputAnalogActionData_t;
    { Raw gyro / accelerometer data of a controller that has them (rotation
      quaternion, acceleration, angular velocity); zeros otherwise. }
    function GetMotionData(inputHandle: InputHandle_t): InputMotionData_t;
    { Calls Init(AExplicitlyCallRunFrame) and remembers it, so the plug-in
      calls Shutdown when Steam shuts down (or on @link(Stop)). With @false
      (the default) Steam's callback pump runs RunFrame; with @true the game
      calls RunFrame itself before reading the actions each frame.
      @returns(@true when ISteamInput started.)
      @raises(ESteamworksError when the loaded library lacks the Init export.) }
    function Start(AExplicitlyCallRunFrame: Boolean = False): Boolean;
    { Calls Shutdown if @link(Start) succeeded; harmless otherwise. }
    procedure Stop;
    { Handles of the connected controllers (at most STEAM_INPUT_MAX_COUNT);
      empty until @link(Start) and the first RunFrame. }
    function ConnectedControllers: TSteamInputHandleArray;
    { The origins (buttons) bound to a digital action in an action set,
      for glyphs and "press X" texts (GetStringForActionOrigin). }
    function DigitalActionOrigins(inputHandle: InputHandle_t; actionSetHandle: InputActionSetHandle_t;
      digitalActionHandle: InputDigitalActionHandle_t): TSteamInputOriginArray;
    { The origins (sticks, triggers, pads) bound to an analog action. }
    function AnalogActionOrigins(inputHandle: InputHandle_t; actionSetHandle: InputActionSetHandle_t;
      analogActionHandle: InputAnalogActionHandle_t): TSteamInputOriginArray;
    { @true between a successful @link(Start) and @link(Stop). }
    property Started: Boolean read FStarted;
//@initialize
  F_GetDigitalActionData := nil;
  F_GetAnalogActionData := nil;
  F_GetMotionData := nil;
  FStarted := False;
  Bind(F_GetDigitalActionData, PFX + 'GetDigitalActionData', False);
  Bind(F_GetAnalogActionData, PFX + 'GetAnalogActionData', False);
  Bind(F_GetMotionData, PFX + 'GetMotionData', False);
//@finalize
  Stop;
  F_GetDigitalActionData := nil;
  F_GetAnalogActionData := nil;
  F_GetMotionData := nil;
//@implementation
function TSteamInputPlugin.GetDigitalActionData(inputHandle: InputHandle_t;
  digitalActionHandle: InputDigitalActionHandle_t): InputDigitalActionData_t;
{$if not (defined(STEAMWORKS_CPU386) and not defined(STEAMWORKS_WINDOWS))}
var
  R: UInt16;
{$endif}
begin
  Need(Assigned(F_GetDigitalActionData), 'GetDigitalActionData');
{$if defined(STEAMWORKS_CPU386) and not defined(STEAMWORKS_WINDOWS)}
  Result := F_GetDigitalActionData(FIntf, inputHandle, digitalActionHandle);
{$else}
  R := F_GetDigitalActionData(FIntf, inputHandle, digitalActionHandle);
  Move(R, Result, SizeOf(Result));
{$endif}
end;

function TSteamInputPlugin.GetAnalogActionData(inputHandle: InputHandle_t;
  analogActionHandle: InputAnalogActionHandle_t): InputAnalogActionData_t;
{$if not defined(STEAMWORKS_WINDOWS) and not defined(STEAMWORKS_CPU386)}
var
  R: TSteamInputRet16;
{$endif}
begin
  Need(Assigned(F_GetAnalogActionData), 'GetAnalogActionData');
{$if defined(STEAMWORKS_WINDOWS)}
  FillChar(Result, SizeOf(Result), 0);
  F_GetAnalogActionData(@Result, FIntf, inputHandle, analogActionHandle);
{$elseif defined(STEAMWORKS_CPU386)}
  Result := F_GetAnalogActionData(FIntf, inputHandle, analogActionHandle);
{$else}
  R := F_GetAnalogActionData(FIntf, inputHandle, analogActionHandle);
  Move(R, Result, SizeOf(Result));
{$endif}
end;

function TSteamInputPlugin.GetMotionData(inputHandle: InputHandle_t): InputMotionData_t;
begin
  Need(Assigned(F_GetMotionData), 'GetMotionData');
  // same test as the TSteamInputMotionFn declarations: procedure with an
  // explicit result pointer everywhere except ARM64 (X8) and Linux32
{$if defined(STEAMWORKS_WINDOWS) or not (defined(STEAMWORKS_CPUARM64) or defined(STEAMWORKS_CPU386))}
  FillChar(Result, SizeOf(Result), 0);
  F_GetMotionData(@Result, FIntf, inputHandle);
{$else}
  Result := F_GetMotionData(FIntf, inputHandle);
{$endif}
end;

function TSteamInputPlugin.Start(AExplicitlyCallRunFrame: Boolean): Boolean;
begin
  if FStarted then
    Exit(True);
  FStarted := Init(AExplicitlyCallRunFrame);
  if FStarted then
    Log(sllInfo, 'ISteamInput started (explicit RunFrame: ' + BoolToStr(AExplicitlyCallRunFrame, True) + ')')
  else
    Log(sllWarning, 'ISteamInput Init returned false');
  Result := FStarted;
end;

procedure TSteamInputPlugin.Stop;
begin
  if not FStarted then
    Exit;
  FStarted := False;
  if (FIntf <> nil) and Assigned(FFlat.Shutdown) then
  begin
    FFlat.Shutdown(FIntf);
    Log(sllInfo, 'ISteamInput shut down');
  end;
end;

function TSteamInputPlugin.ConnectedControllers: TSteamInputHandleArray;
var
  Buf: array[0..STEAM_INPUT_MAX_COUNT - 1] of InputHandle_t;
  N, I: Integer;
begin
  Result := nil;
  FillChar(Buf, SizeOf(Buf), 0);
  N := GetConnectedControllers(@Buf[0]);
  if N > STEAM_INPUT_MAX_COUNT then
    N := STEAM_INPUT_MAX_COUNT;
  if N <= 0 then
    Exit;
  SetLength(Result, N);
  for I := 0 to N - 1 do
    Result[I] := Buf[I];
end;
